## Assessment and recovery decision

Moving cases earlier helped some boundaries but did not establish reliable
one-repair recovery. A larger list of cases is not by itself a root-cause fix.
The persistent coding problem is loss of value provenance: generated code treats
an inherited default and an explicit choice alike, then repairs one precedence
case by changing both. Broad case groups also leave individual combinations
unexamined even when their evidence is marked tested.

Keep the live PR paused. Do not automatically increase retries, rerun the same
experiment until it passes, or describe these results as sufficient prevention.
The next useful investigation is whether the task contract can state a consistent
precedence/provenance rule, with explicit inherited-versus-selected examples,
before implementation. Conflicting expectations should become a clarification
instead of alternating fixes. That investigation should precede further model
spending or a routing change; independent review must remain the final gate.

No CLI replan action or live reconciliation was performed because the experiment
did not pass its prerequisite. The archived manual delta, live files, task history,
and consumed repair allowance remain intact.

"""Bounded behavioral cases and evidence identities for Owner verification.

Evidence is a provider report, not a backend attestation that commands ran.
The backend enforces coverage, consistency and identity; review stays independent.
"""
import copy
import re
from .common import ApiError, digest

MAX_CASES = 8


def cases(value, *, code='invalid_result'):
    if not isinstance(value, list) or len(value) > MAX_CASES:
        raise ApiError(code, 'Supply at most eight focused verification cases per task.')
    seen = set()
    for case in value:
        if (not isinstance(case, dict) or not isinstance(case.get('id'), str)
                or not re.fullmatch(r'[A-Za-z0-9_-]{1,64}', case['id']) or case['id'] in seen
                or case.get('intent') not in ('preserve', 'change')
                or any(not isinstance(case.get(k), str) or not case[k].strip()
                       for k in ('scenario', 'expected', 'basis'))):
            raise ApiError(code, 'Verification cases need unique IDs, intent, scenario, expectation and supporting basis.')
        seen.add(case['id'])
    return copy.deepcopy(value)


def task_cases(task):
    return copy.deepcopy(task.get('verification_cases', []))


def combined(tasks):
    return [{**case, 'id': task['id'] + '/' + case['id']} for task in tasks
            if task['status'] == 'complete' for case in task_cases(task)]


def bind(report, required, tree):
    """Validate new evidence and bind it to cases and the actual source checkpoint."""
    supplied = report.get('case_results')
    if required or supplied is not None:
        if not isinstance(supplied, list) or any(not isinstance(r, dict) for r in supplied):
            raise ApiError('invalid_result', 'Report verification evidence for every required case.')
        if any(not isinstance(r.get('id'), str) for r in supplied):
            raise ApiError('invalid_result', 'Case evidence IDs must be strings.')
        if sorted(r.get('id', '') for r in supplied) != sorted(c['id'] for c in required):
            raise ApiError('invalid_result', 'Case evidence must cover each required case exactly once.')
        for row in supplied:
            if (row.get('status') not in ('tested', 'source_traced', 'failed', 'unavailable')
                    or not isinstance(row.get('evidence'), str) or not row['evidence'].strip()):
                raise ApiError('invalid_result', 'Every case requires a supported status and concrete evidence.')
        if any(r['status'] in ('failed', 'unavailable') for r in supplied) and report.get('complete') and not report.get('blocking'):
            raise ApiError('invalid_result', 'Failed or unavailable case evidence cannot support a passed outcome.')
        if any(r['status'] == 'unavailable' for r in supplied) and report.get('assessment') == 'changes_requested':
            raise ApiError('invalid_result', 'Unavailable verification evidence requires an incomplete assessment.')
    if required or supplied is not None:
        report.update(case_contract=digest(required), case_tree=tree)
    return report


def implementation(task, report, tree):
    """A fixer may add cases, never rewrite existing requirements during a repair."""
    previous = task_cases(task)
    proposed = cases(report.get('verification_cases', []))
    by_id = {c['id']: c for c in previous}
    for case in proposed:
        if case['id'] in by_id and by_id[case['id']] != case:
            raise ApiError('invalid_result', 'Repair changed an existing verification case; clarify or explicitly replan.')
        by_id[case['id']] = case
    required = cases(list(by_id.values()))
    bind(report, required, tree)
    if required:
        task['verification_cases'] = required
        task['case_contract_version'] = 1
    return report


def matches(report, required, tree):
    if not required and 'case_contract' not in report:
        return True  # Historical reports did not supply case-level evidence.
    if report.get('case_contract') != digest(required) or report.get('case_tree') != tree:
        return False
    try:
        bind(copy.deepcopy(report), required, tree)
        return True
    except (ApiError, TypeError):
        return False


def verified(report, required, tree):
    if not required and 'case_contract' not in report:
        return True
    return (matches(report, required, tree) and report.get('complete') is True
            and not report.get('blocking')
            and all(r['status'] in ('tested', 'source_traced') for r in report.get('case_results', [])))


def current(batch):
    for task in batch.get('tasks', []):
        if task['status'] == 'complete' and not verified(task.get('review', {}), task_cases(task), task.get('after', {}).get('tree')):
            return False
    return verified(batch.get('review', {}), combined(batch.get('tasks', [])), batch.get('final_tree'))


def require_current(batch):
    if not current(batch):
        raise ApiError('stale_verification', 'Verification cases or source changed; reverify before committing or publishing.')

"""Review configuration, including read-only migration of legacy manifests."""
from .common import ApiError, read_json
from .state import Session, discover

DEFAULTS = {"reviewers": ["claude", "codex"], "processing_provider": "codex",
            "claude_model": "opus", "codex_model": "gpt-6.1-sol",
            "claude_effort": "high", "codex_effort": "high", "synthesis_tests": "focused",
            "suggested_changes": False}

# Manifests predating explicit configuration used these settings. New defaults
# must not silently change their providers or invalidate resumable work.
LEGACY_DEFAULTS = {**DEFAULTS, "processing_provider": "claude",
                   "claude_model": "fable", "codex_model": "gpt-6-astra"}

ECONOMICAL = {"reviewers": ["claude", "codex"], "processing_provider": "codex",
              "claude_model": "haiku", "codex_model": "gpt-6.1-terra",
              "claude_effort": "high", "codex_effort": "high"}
EXPENSIVE = {**ECONOMICAL, "claude_model": "fable", "codex_model": "gpt-6-astra"}
PRESETS = {"economical": ECONOMICAL, "expensive": EXPENSIVE}


def from_manifest(manifest):
    models = manifest.get("models", {})
    result = dict(LEGACY_DEFAULTS)
    result["synthesis_tests"] = manifest.get("phases", {}).get("synthesis", {}).get("testing_policy", "focused")
    for provider in ("claude", "codex"):
        result[provider + "_model"] = models.get(provider) or LEGACY_DEFAULTS[provider + "_model"]
        result[provider + "_effort"] = models.get(provider + "_effort", models.get("effort", "high"))
    result.update(manifest.get("configuration", {}))
    return result


def validate(settings):
    result = {k: settings.get(k, default) for k, default in DEFAULTS.items()}
    if type(result['suggested_changes']) is not bool:
        raise ApiError('invalid_request', 'suggested_changes must be a boolean.')
    reviewers = result["reviewers"]
    if (not isinstance(reviewers, list) or not reviewers or
            any(p not in ("claude", "codex") for p in reviewers) or len(set(reviewers)) != len(reviewers)):
        raise ApiError("invalid_request", "Choose Claude, Codex, or both reviewers.")
    result["reviewers"] = [p for p in ("claude", "codex") if p in reviewers]
    if len(reviewers) == 1:
        result["processing_provider"] = reviewers[0]
    if result["processing_provider"] not in reviewers:
        raise ApiError("invalid_request", "The processing provider must be an enabled reviewer.")
    if result["synthesis_tests"] not in ("focused", "full"):
        raise ApiError("invalid_request", "synthesis_tests must be focused or full.")
    for key in ("claude_model", "codex_model", "claude_effort", "codex_effort"):
        value = result[key]
        if not isinstance(value, str) or not value.strip() or value.startswith("-") or any(ord(c) < 32 for c in value):
            raise ApiError("invalid_request", "Invalid model or effort option.")
    return result


def initial_settings(request):
    """Expand new-review defaults consistently for CLI and API entry points."""
    settings = dict(DEFAULTS)
    defaults = request.get("defaults", {})
    settings.update({k: v for k, v in defaults.items() if k in DEFAULTS and v is not None})
    for name in PRESETS:
        if type(request.get(name, False)) is not bool:
            raise ApiError('invalid_request', f'{name} must be a boolean.')
    selected = [name for name in PRESETS if request.get(name)]
    if len(selected) > 1:
        raise ApiError('invalid_request', 'Choose economical or expensive, not both.')
    if selected:
        settings.update(PRESETS[selected[0]])
    return settings


def resolve(request):
    settings = initial_settings(request)
    session_path = request.get("session")
    if not session_path and request.get("pr"):
        matches = discover({**request, "pr": str(request["pr"]).rstrip("/").split("/")[-1]})["sessions"]
        if len(matches) > 1:
            paths = "\n".join("  " + match["session"] for match in matches)
            raise ApiError("ambiguous_session", "Multiple saved runs match this checkout and PR; select one with --resume RUN_DIRECTORY:\n" + paths)
        if matches:
            session_path = matches[0]["session"]
    if session_path:
        session = Session({"session": session_path})
        manifest = read_json((session.pointer(".pending-review") or session.path) / "manifest.json", {})
        settings.update(from_manifest(manifest))
    settings.update({k: request[k] for k in DEFAULTS if k in request})
    return {"settings": validate(settings), "session": str(session_path) if session_path else None}
